# 读取文件

层  x  y

```java
FileHandle file = Gdx.files.internal("data/txt/Level/pos_" + level + ".txt");
String[] s = file.readString().replaceAll("\r", "").split("]\n");
String[] ss = s[0].substring(1).split(", ");
for (String value : ss) {
    this.layerArray.add(Integer.valueOf(value));
}
String[] ss2 = s[1].substring(1).split(", ");
for (String value2 : ss2) {
    this.xArray.add(Integer.valueOf(value2));
}
String[] ss3 = s[2].substring(1).split(", ");
for (String value3 : ss3) {
    this.yArray.add(Integer.valueOf(value3));
}

```

## 初始化麻将

```java
MahjongImage[] mahjongs = new MahjongImage[this.layerArray.size];
    for (int i = 0; i < this.layerArray.size; i++) {
        mahjongs[i] = new MahjongImage(this.layerArray.get(i).intValue(), this.xArray.get(i).intValue(), this.yArray.get(i).intValue(), this.myGame.polygonBatch);
        this.mahjongArray.add(mahjongs[i]);
        if (i > 0) {
            Iterator<MahjongImage> it = this.mahjongArray.iterator();
            while (it.hasNext()) {
                MahjongImage existingMahjong = it.next();
                int delta_layer = mahjongs[i].layer - existingMahjong.layer;
                int delta_x = Math.abs(mahjongs[i].x - existingMahjong.x);
                int delta_y = Math.abs(mahjongs[i].y - existingMahjong.y);
                if (delta_layer == 0 && delta_x == 2 && delta_y <= 1) {
                    existingMahjong.addLimit(mahjongs[i]);
                    mahjongs[i].addBeLimited(existingMahjong);
                    mahjongs[i].addLimit(existingMahjong);
                    existingMahjong.addBeLimited(mahjongs[i]);
                } else if (delta_layer >= 1 && delta_x <= 1 && delta_y <= 1) {
                    existingMahjong.addLimit(mahjongs[i]);
                    mahjongs[i].addBeLimited(existingMahjong);
                }
            }
        }
    }
```
